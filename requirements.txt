import pandas as pd
import plotly.express as px
import streamlit as st

from scraper import fetch_many
from storage import authenticate, create_user, get_saved_funds, set_saved_funds

st.set_page_config(page_title="BES Fon Takip", page_icon="📈", layout="wide")

DEFAULT_FUNDS = ["AMZ", "AGH", "GHO", "FFC", "AZY", "AZL", "ALI", "AUA"]


def pct_fmt(x):
    if pd.isna(x):
        return ""
    return f"{x:.2f}%".replace(".", ",")


def price_fmt(x):
    if pd.isna(x):
        return ""
    return f"{x:.6f}".replace(".", ",")


def login_panel():
    st.title("📈 BES Fon Takip Uygulaması")
    tab_login, tab_register = st.tabs(["Giriş", "Yeni kullanıcı"])

    with tab_login:
        username = st.text_input("Kullanıcı adı", key="login_user")
        password = st.text_input("Şifre", type="password", key="login_password")
        if st.button("Giriş yap", type="primary"):
            if authenticate(username, password):
                st.session_state["logged_in"] = True
                st.session_state["username"] = username.lower().strip()
                st.rerun()
            else:
                st.error("Kullanıcı adı veya şifre hatalı.")

    with tab_register:
        new_user = st.text_input("Yeni kullanıcı adı", key="new_user")
        new_password = st.text_input("Yeni şifre", type="password", key="new_password")
        if st.button("Kullanıcı oluştur"):
            if not new_user.strip() or not new_password:
                st.warning("Kullanıcı adı ve şifre giriniz.")
            else:
                try:
                    create_user(new_user, new_password)
                    set_saved_funds(new_user, DEFAULT_FUNDS)
                    st.success("Kullanıcı oluşturuldu. Şimdi giriş yapabilirsiniz.")
                except ValueError as exc:
                    st.error(str(exc))


def parse_codes(text: str):
    return [x.strip().upper() for x in text.replace(";", ",").replace(" ", ",").split(",") if x.strip()]


if not st.session_state.get("logged_in"):
    login_panel()
    st.stop()

username = st.session_state["username"]
st.sidebar.success(f"Giriş yapan kullanıcı: {username}")
if st.sidebar.button("Çıkış"):
    st.session_state.clear()
    st.rerun()

st.title("BES Fon Getirileri")
st.caption("Veriler besfongetirileri.com fon kartı sayfalarından çekilir. Bu uygulama yatırım tavsiyesi değildir.")

saved = get_saved_funds(username)
if not saved:
    saved = DEFAULT_FUNDS
    set_saved_funds(username, saved)

with st.sidebar:
    st.header("Kaydettiğim fonlar")
    saved_text = st.text_area("Fon kodları", value=", ".join(saved), height=120)
    if st.button("Listeyi kaydet"):
        set_saved_funds(username, parse_codes(saved_text))
        st.success("Fon listen kaydedildi.")
        st.rerun()

    st.divider()
    st.header("Diğer fon sorgula")
    query_text = st.text_input("Fon kodu veya kodları", placeholder="Örn: AMZ, AGH")

main_tab, query_tab = st.tabs(["Kaydettiğim fonlar", "Fon sorgula"])

with main_tab:
    st.subheader("Kayıtlı fon performansım")
    col1, col2 = st.columns([1, 3])
    with col1:
        refresh = st.button("Verileri yenile", type="primary")

    with st.spinner("Fon verileri çekiliyor..."):
        rows, errors = fetch_many(saved)

    if errors:
        st.warning("Bazı fonlar çekilemedi: " + ", ".join([e["fon_kodu"] for e in errors]))
        with st.expander("Hata detayları"):
            st.json(errors)

    if rows:
        df = pd.DataFrame(rows)
        display_cols = [
            "fon_kodu", "fon_adi", "son_fiyat", "getiri_1_ay", "getiri_3_ay",
            "getiri_6_ay", "getiri_1_yil", "fon_turu", "risk_degeri", "cekilme_zamani"
        ]
        display_cols = [c for c in display_cols if c in df.columns]
        st.dataframe(
            df[display_cols].style.format({
                "son_fiyat": price_fmt,
                "getiri_1_ay": pct_fmt,
                "getiri_3_ay": pct_fmt,
                "getiri_6_ay": pct_fmt,
                "getiri_1_yil": pct_fmt,
                "risk_degeri": lambda x: "" if pd.isna(x) else f"{x:.2f}".replace(".", ","),
            }),
            use_container_width=True,
            hide_index=True,
        )

        metric = st.selectbox(
            "Grafikte gösterilecek getiri",
            ["getiri_1_ay", "getiri_3_ay", "getiri_6_ay", "getiri_1_yil"],
            format_func=lambda x: {
                "getiri_1_ay": "1 Ay",
                "getiri_3_ay": "3 Ay",
                "getiri_6_ay": "6 Ay",
                "getiri_1_yil": "1 Yıl",
            }[x],
        )
        chart_df = df.dropna(subset=[metric]).sort_values(metric, ascending=False)
        fig = px.bar(chart_df, x="fon_kodu", y=metric, hover_data=["fon_adi"], title="Fon Getiri Karşılaştırması")
        st.plotly_chart(fig, use_container_width=True)

        csv = df.to_csv(index=False).encode("utf-8-sig")
        st.download_button("CSV indir", csv, "bes_fon_getirileri.csv", "text/csv")
    else:
        st.info("Kayıtlı fon bulunamadı. Sol menüden fon kodu ekleyin.")

with query_tab:
    st.subheader("Diğer fonları sorgula")
    codes = parse_codes(query_text)
    if st.button("Sorgula", key="query_button"):
        if not codes:
            st.warning("En az bir fon kodu giriniz.")
        else:
            with st.spinner("Sorgulanıyor..."):
                rows, errors = fetch_many(codes)
            if rows:
                qdf = pd.DataFrame(rows)
                st.dataframe(qdf, use_container_width=True, hide_index=True)
                add_codes = st.multiselect("Kaydedilecek fonları seç", qdf["fon_kodu"].tolist())
                if st.button("Seçilenleri kaydet"):
                    current = get_saved_funds(username)
                    set_saved_funds(username, current + add_codes)
                    st.success("Seçilen fonlar kayıtlı listene eklendi.")
            if errors:
                st.error("Bazı fonlar bulunamadı veya çekilemedi.")
                st.json(errors)
